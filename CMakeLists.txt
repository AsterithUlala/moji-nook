cmake_minimum_required(VERSION 3.21)
project(MojiNook VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network Sql Test Multimedia)
if(UNIX AND NOT APPLE)
  find_package(LayerShellQt REQUIRED)
endif()
add_library(moji_nook_core src/core.cpp src/picker.cpp src/store.cpp src/stats.cpp src/sync.cpp src/analytics.cpp src/anki.cpp src/progression.cpp)
target_include_directories(moji_nook_core PUBLIC src)
target_link_libraries(moji_nook_core PUBLIC Qt6::Network Qt6::Sql Qt6::Gui)
include(cmake/SpeechBundle.cmake)
set(MOJI_NOOK_UI_SOURCES assets/brand/brand.qrc assets/print-ink/print-ink.qrc src/progression_accessibility.cpp src/progression_widgets.cpp src/progression_popup.cpp src/display_preview.cpp src/voice_settings.cpp src/ui.cpp src/card.cpp src/practice_clock.cpp src/style.cpp
    src/dashboard.cpp src/preview.cpp src/demo.cpp src/charts.cpp src/feedback.cpp src/modes.cpp src/anki_widgets.cpp)
add_executable(moji_nook WIN32 src/main.cpp ${MOJI_NOOK_UI_SOURCES})
target_link_libraries(moji_nook PRIVATE moji_nook_core moji_nook_speech Qt6::Widgets Qt6::Multimedia)
if(UNIX AND NOT APPLE)
  target_link_libraries(moji_nook PRIVATE LayerShellQt::Interface)
  target_compile_definitions(moji_nook PRIVATE MOJI_NOOK_LAYER_SHELL)
endif()
if(WIN32)
  target_link_libraries(moji_nook PRIVATE user32)
endif()
enable_testing()
add_executable(moji_nook_anki_tests tests/anki_test.cpp)
target_link_libraries(moji_nook_anki_tests PRIVATE moji_nook_core Qt6::Widgets Qt6::Test)
add_test(NAME anki COMMAND moji_nook_anki_tests -platform offscreen)
add_executable(moji_nook_sync_tests tests/sync_test.cpp)
target_link_libraries(moji_nook_sync_tests PRIVATE moji_nook_core Qt6::Test)
add_test(NAME sync COMMAND moji_nook_sync_tests)
add_executable(moji_nook_tests tests/core_test.cpp)
target_link_libraries(moji_nook_tests PRIVATE moji_nook_core Qt6::Test)
add_test(NAME core COMMAND moji_nook_tests)
add_executable(moji_nook_progression_tests tests/progression_test.cpp)
target_link_libraries(moji_nook_progression_tests PRIVATE moji_nook_core Qt6::Test)
add_test(NAME progression COMMAND moji_nook_progression_tests)
add_executable(moji_nook_ui_tests tests/ui_test.cpp ${MOJI_NOOK_UI_SOURCES})
target_link_libraries(moji_nook_ui_tests PRIVATE moji_nook_core moji_nook_speech Qt6::Widgets Qt6::Test Qt6::Multimedia)
if(WIN32)
  target_link_libraries(moji_nook_ui_tests PRIVATE user32)
endif()
add_test(NAME ui COMMAND moji_nook_ui_tests -platform offscreen)
if(UNIX AND NOT APPLE)
  add_executable(moji_nook_desktop_check tests/desktop_check.cpp ${MOJI_NOOK_UI_SOURCES})
  target_link_libraries(moji_nook_desktop_check PRIVATE moji_nook_core moji_nook_speech Qt6::Widgets Qt6::Test Qt6::Multimedia LayerShellQt::Interface)
  target_compile_definitions(moji_nook_desktop_check PRIVATE MOJI_NOOK_LAYER_SHELL)
endif()
set_target_properties(moji_nook PROPERTIES OUTPUT_NAME "moji-nook")
install(TARGETS moji_nook RUNTIME DESTINATION bin)
if(UNIX AND NOT APPLE)
  install(FILES assets/moji-nook.desktop DESTINATION share/applications)
  install(FILES assets/moji-nook.svg DESTINATION share/icons/hicolor/scalable/apps)
  foreach(icon_size 16 22 24 32 48 64 128 256 512)
    install(FILES assets/brand/moji-nook-${icon_size}.png
            DESTINATION share/icons/hicolor/${icon_size}x${icon_size}/apps
            RENAME moji-nook.png)
  endforeach()
endif()
add_executable(moji_nook_performance EXCLUDE_FROM_ALL tests/performance.cpp ${MOJI_NOOK_UI_SOURCES})
target_link_libraries(moji_nook_performance PRIVATE moji_nook_core moji_nook_speech Qt6::Widgets Qt6::Multimedia)
if(WIN32)
  target_link_libraries(moji_nook_performance PRIVATE user32)
endif()

add_executable(moji_nook_speech_tests tests/speech_test.cpp)
target_link_libraries(moji_nook_speech_tests PRIVATE moji_nook_speech Qt6::Widgets Qt6::Test)
add_test(NAME speech COMMAND moji_nook_speech_tests -platform offscreen)
